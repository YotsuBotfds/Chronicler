"""Copy evidence stdout logs without changing compared simulation directories.

Open descriptors survive the comparator's unlink of passing resume logs.
No simulation/native import or checkpoint loading occurs in this collector.
"""
import argparse
import hashlib
import json
from pathlib import Path
import time


def main():
    p = argparse.ArgumentParser(description=__doc__)
    p.add_argument("--run", required=True)
    p.add_argument("--case", required=True)
    p.add_argument("--max-seconds", type=float, required=True)
    a = p.parse_args()
    run, started = Path(a.run), time.monotonic()
    out = run / "retained-logs"
    out.mkdir(exist_ok=False)
    held = {}
    try:
        while True:
            for path in sorted((run / a.case).glob("*.log")):
                if path.name not in held:
                    held[path.name] = path.open("rb")
            for name, stream in held.items():
                stream.seek(0)
                (out / name).write_bytes(stream.read())
            done = (run / "report.json").is_file() or (run / "failure.json").is_file()
            if done or time.monotonic() - started > a.max_seconds:
                receipt = {"done": done, "seconds": time.monotonic() - started,
                           "collector_sha256": hashlib.sha256(Path(__file__).read_bytes()).hexdigest(),
                           "files": {name: {"bytes": (out / name).stat().st_size,
                                            "sha256": hashlib.sha256((out / name).read_bytes()).hexdigest()}
                                     for name in held},
                           "source": "stdout log descriptors; no compared outputs modified"}
                (run / "retained-logs.json").write_text(json.dumps(receipt, indent=2) + "\n")
                return 0 if done else 1
            time.sleep(.2)
    finally:
        for stream in held.values():
            stream.close()


if __name__ == "__main__":
    raise SystemExit(main())
